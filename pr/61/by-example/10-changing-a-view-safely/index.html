Changing a view safely¶

A message outlives the code that sent it. This chapter shows which changes old polls survive, and how to answer clicks on the ones that do not.

The free change¶

Add a field at the end of the class, with a default. This poll gets a note under its question.

@risa.register(name="poll", version=2, state=risa.InStore(ttl=WEEK))
class Poll(risa.View):
    question: str
    yes: int = 0
    no: int = 0
    voters: list[int] = msgspec.field(default_factory=list[int])
    note: str = "One vote each."

    def render(self) -> ui.Layout:
        return ui.Container(
            ui.TextDisplay(f"## {self.question}"),
            ui.TextDisplay(self.note),
            ui.Row(
                ui.Button(self.vote_yes, label=f"Yes: {self.yes}"),
                ui.Button(self.vote_no, label=f"No: {self.no}"),
            ),
        )

An old poll keeps working. Its next click fills note with the default, rerender() draws it, and the votes are still there. In a message the new field takes room too: with this note, the poll from Where state lives has room for ten voters, and one that already holds ten or more fails on its next vote. A field anywhere else, or without a default, is not free.

Try it.

Send a poll with the Redis file from Stores and TTL and stop the bot. Start the file below on the same Redis and click Yes on the old poll. It counts your vote and gains the note.

Full file 
"""The poll gains a field with a default, and polls already sent keep working."""

from __future__ import annotations

import os

import hikari
import msgspec
import redis.asyncio

import risa
from risa import ui
from risa.store.redis import RedisStore

WEEK = 7 * 24 * 60 * 60


@risa.register(name="poll", version=2, state=risa.InStore(ttl=WEEK))
class Poll(risa.View):
    question: str
    yes: int = 0
    no: int = 0
    voters: list[int] = msgspec.field(default_factory=list[int])
    note: str = "One vote each."

    def render(self) -> ui.Layout:
        return ui.Container(
            ui.TextDisplay(f"## {self.question}"),
            ui.TextDisplay(self.note),
            ui.Row(
                ui.Button(self.vote_yes, label=f"Yes: {self.yes}"),
                ui.Button(self.vote_no, label=f"No: {self.no}"),
            ),
        )

    @risa.handler
    async def vote_yes(self, ctx: risa.ComponentContext) -> None:
        if await self._first_vote(ctx):
            self.yes += 1
            await ctx.rerender()

    @risa.handler
    async def vote_no(self, ctx: risa.ComponentContext) -> None:
        if await self._first_vote(ctx):
            self.no += 1
            await ctx.rerender()

    async def _first_vote(self, ctx: risa.ComponentContext) -> bool:
        if ctx.user.id in self.voters:
            await ctx.respond("You already voted.", ephemeral=True)
            return False
        self.voters.append(ctx.user.id)
        return True

    @classmethod
    async def on_state_missing(cls, ctx: risa.ComponentContext) -> None:
        await ctx.edit(ui.TextDisplay("This poll has expired."))


def main() -> None:
    bot = hikari.GatewayBot(
        os.environ["DISCORD_TOKEN"],
        intents=hikari.Intents.ALL_UNPRIVILEGED | hikari.Intents.MESSAGE_CONTENT,
    )
    connection = redis.asyncio.Redis(host="localhost")
    client = risa.client_from_app(bot, store=RedisStore(connection))
    client.add_view(Poll)

    @bot.listen()
    async def on_stopped(_: hikari.StoppedEvent) -> None:
        await connection.aclose()

    @bot.listen()
    async def on_message(event: hikari.GuildMessageCreateEvent) -> None:
        if event.content == "!poll":
            rendered = await client.build(Poll(question="Pizza on Friday?"))
            await rendered.send_to(bot.rest, event.channel_id)

    bot.run()


if __name__ == "__main__":
    main()

A change that is not free¶

The poll should now let people change their mind, so a vote has to record who chose what. yes and no become lists, and voters goes. An old record cannot say who chose what, so the view's version goes up.

@risa.register(name="poll", version=3, state=risa.InStore(ttl=WEEK))
class Poll(risa.View):
    question: str
    yes: list[int] = msgspec.field(default_factory=list[int])
    no: list[int] = msgspec.field(default_factory=list[int])
    note: str = "Change your mind whenever you like."

version is part of the route in each button's id. A click on an old poll still names version 2, and no view answers to that, so the click fails in Discord. on_outdated does not run, because it belongs to the class at version 3.

  1. MondayVersion 2 sends a pollIts buttons route to version 2 and hold the record's key.
  2. TuesdayYou release version 3New polls route to version 3. The old message stays as it was.
  3. WednesdaySomeone clicks the old pollThe click fails in Discord, unless a class at version 2 answers it.
  4. A week after the last click that read itThe record expiresThe message stays, and a later click finds no record.

Answering old polls¶

To answer those clicks, keep a class at the old version, with the old fields, and add both to the client.

@risa.register(name="poll", version=2, state=risa.InStore(ttl=WEEK))
class OldPoll(risa.View):
    question: str
    yes: int = 0
    no: int = 0
    voters: list[int] = msgspec.field(default_factory=list[int])
    note: str = ""

client.add_view(Poll)
client.add_view(OldPoll)

Keep the old handlers too, with the same ids, versions and arguments, and let their bodies say the poll is over. Such a rejection handler routes like any other and logs nothing, and self holds the old record.

@risa.handler
async def vote_yes(self, ctx: risa.ComponentContext) -> None:
    await self._ended(ctx)

@risa.handler
async def vote_no(self, ctx: risa.ComponentContext) -> None:
    await self._ended(ctx)

async def _ended(self, ctx: risa.ComponentContext) -> None:
    text = f"{self.question} It ended with {self.yes} yes and {self.no} no."
    await ctx.edit(ui.TextDisplay(text))

on_outdated¶

View.on_outdated answers a click that reaches a view but cannot run: its handler has gone, or the record no longer fits the fields. ctx has no view, so ctx.rerender() raises NoViewError. By default it does nothing, and the click fails in Discord.

@classmethod
async def on_outdated(cls, ctx: risa.ComponentContext) -> None:
    await ctx.respond("This button no longer works.", ephemeral=True)

After a click on a retired button

Risa BotAPPToday at 12:00

This button no longer works.

Only you can see this·Dismiss message

If you do not need the old record, a class at the old version needs no fields, no handlers and no state=: give it this method, and every old button lands there.

A handler has a version of its own too. This is what each change does:

What you changed A click on an old message risa logs
Appended a field with a default Works, while the state still fits Nothing, or ERROR once it no longer fits
Raised the view's version Fails in Discord, unless a class at the old version stays DEBUG
Changed the fields or moved the view to a store, with no new version on_outdated ERROR
Changed a handler's arguments, with no new version on_outdated ERROR
Deleted a handler, or raised its version and kept no method at the old one on_outdated WARNING, or DEBUG once you override it

A record that is gone goes to on_state_missing instead.

"""A vote now records who chose what, so the view's version goes up. Version 2 stays to answer old polls."""

from __future__ import annotations

import os

import hikari
import msgspec
import redis.asyncio

import risa
from risa import ui
from risa.store.redis import RedisStore

WEEK = 7 * 24 * 60 * 60


@risa.register(name="poll", version=3, state=risa.InStore(ttl=WEEK))
class Poll(risa.View):
    question: str
    yes: list[int] = msgspec.field(default_factory=list[int])
    no: list[int] = msgspec.field(default_factory=list[int])
    note: str = "Change your mind whenever you like."

    def render(self) -> ui.Layout:
        return ui.Container(
            ui.TextDisplay(f"## {self.question}"),
            ui.TextDisplay(self.note),
            ui.Row(
                ui.Button(self.vote_yes, label=f"Yes: {len(self.yes)}"),
                ui.Button(self.vote_no, label=f"No: {len(self.no)}"),
            ),
        )

    @risa.handler
    async def vote_yes(self, ctx: risa.ComponentContext) -> None:
        self._take_back(ctx.user.id)
        self.yes.append(ctx.user.id)
        await ctx.rerender()

    @risa.handler
    async def vote_no(self, ctx: risa.ComponentContext) -> None:
        self._take_back(ctx.user.id)
        self.no.append(ctx.user.id)
        await ctx.rerender()

    def _take_back(self, voter: int) -> None:
        self.yes = [other for other in self.yes if other != voter]
        self.no = [other for other in self.no if other != voter]

    @classmethod
    async def on_outdated(cls, ctx: risa.ComponentContext) -> None:
        await ctx.respond("This button no longer works.", ephemeral=True)

    @classmethod
    async def on_state_missing(cls, ctx: risa.ComponentContext) -> None:
        await ctx.edit(ui.TextDisplay("This poll has expired."))


@risa.register(name="poll", version=2, state=risa.InStore(ttl=WEEK))
class OldPoll(risa.View):
    question: str
    yes: int = 0
    no: int = 0
    voters: list[int] = msgspec.field(default_factory=list[int])
    note: str = ""

    @risa.handler
    async def vote_yes(self, ctx: risa.ComponentContext) -> None:
        await self._ended(ctx)

    @risa.handler
    async def vote_no(self, ctx: risa.ComponentContext) -> None:
        await self._ended(ctx)

    async def _ended(self, ctx: risa.ComponentContext) -> None:
        text = f"{self.question} It ended with {self.yes} yes and {self.no} no."
        await ctx.edit(ui.TextDisplay(text))

    @classmethod
    async def on_state_missing(cls, ctx: risa.ComponentContext) -> None:
        await ctx.edit(ui.TextDisplay("This poll has expired."))


def main() -> None:
    bot = hikari.GatewayBot(
        os.environ["DISCORD_TOKEN"],
        intents=hikari.Intents.ALL_UNPRIVILEGED | hikari.Intents.MESSAGE_CONTENT,
    )
    connection = redis.asyncio.Redis(host="localhost")
    client = risa.client_from_app(bot, store=RedisStore(connection))
    client.add_view(Poll)
    client.add_view(OldPoll)

    @bot.listen()
    async def on_stopped(_: hikari.StoppedEvent) -> None:
        await connection.aclose()

    @bot.listen()
    async def on_message(event: hikari.GuildMessageCreateEvent) -> None:
        if event.content == "!poll":
            rendered = await client.build(Poll(question="Pizza on Friday?"))
            await rendered.send_to(bot.rest, event.channel_id)

    bot.run()


if __name__ == "__main__":
    main()
